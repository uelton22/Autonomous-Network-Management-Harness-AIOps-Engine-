Architecture & Implementation Specification: Autonomous Network Harness (AIOps Engine)Document Type: Technical Architecture, Implementation Guide & LLM Context PromptDomain: Network Automation, SSH Session Management, Dynamic TTP Parsing & Intent OrchestrationIntegration Interface: Model Context Protocol (MCP)1. Executive Summary & Core PrinciplesThis specification defines an autonomous network management harness that bridges the gap between non-deterministic Large Language Models (LLMs) and deterministic, multi-vendor network operating system (NOS) CLIs.Core Architectural PillarsZero-Knowledge Progressive Fingerprinting: Discovers vendor, OS, and major version starting only from an IP address without hardcoded credentials or hallucinated discovery logic.Deterministic Canonical Command Registry: Decouples user intent from platform syntax. The LLM translates user requests into canonical intents (e.g., get_system_version), which are resolved via a deterministic matrix rather than raw CLI generation.Adaptive TTP Generation with Self-Healing Loop: Automatically writes, sandboxes, and verifies Template Text Parser (TTP) rules when parsing misses occur in cache.Context Window Optimization via Chunk Sampling: Safely handles multi-megabyte CLI outputs (show tech, display current-configuration) by isolating representative repetition blocks for LLM prompt context.Decoupled Execution via MCP: Isolates the terminal emulation engine (Scraprmiko) in an MCP server providing a single primitive tool: execute_command(host, command, platform_type).2. 6. Module 4: Chunk Sampling Algorithm for Large Raw OutputsWhen parsing outputs with thousands of lines (e.g., full routing tables, dense switchport outputs), the raw data must not be sent in full to the LLM. The harness isolates structural samples.import re

def extract_representative_sample(raw_text: str, max_lines: int = 60) -> str:
    """
    Extracts header, footer, and structural repeating chunks from large CLI outputs.
    Ensures the LLM understands delimiters without exceeding token limits.
    """
    lines = [line for line in raw_text.splitlines() if line.strip()]
    total_lines = len(lines)
    
    if total_lines <= max_lines:
        return raw_text

    head_size = 15
    tail_size = 10
    body_budget = max_lines - (head_size + tail_size)

    # Identifica repetições (linhas que iniciam com interfaces ou prefixos de blocos)
    pattern = re.compile(r"^(\S+\s+\d+|\w+Ethernet\S*|interface\s+\S+|BGP\s+routing\s+table)", re.IGNORECASE)
    matching_indices = [idx for idx, line in enumerate(lines) if pattern.match(line.strip())]

    sampled_body_lines = []
    if matching_indices:
        # Pega as 2 primeiras ocorrências estruturais
        first_match = matching_indices[0]
        sampled_body_lines.extend(lines[first_match : min(first_match + (body_budget // 2), total_lines)])
    else:
        # Fallback para meio do arquivo caso não haja padrão evidente
        mid = total_lines // 2
        sampled_body_lines.extend(lines[mid : mid + body_budget])

    sampled_output = (
        "\n".join(lines[:head_size])
        + "\n\n... [STRUCTURAL SAMPLE TRUNCATED: REPETITIVE BLOCKS OMITTED] ...\n\n"
        + "\n".join(sampled_body_lines)
        + "\n\n... [STRUCTURAL SAMPLE TRUNCATED: FINAL LOGS] ...\n\n"
        + "\n".join(lines[-tail_size:])
    )
    return sampled_output

8. Python Reference Implementation: Engine Coreimport os
import json
from pathlib import Path
from ttp import ttp
from pydantic import ValidationError
from typing import Type, Any
from registry.schemas.system import SystemVersionSchema

class TTPAutoHealingEngine:
    def __init__(self, template_dir: str = "storage/templates", llm_client: Any = None):
        self.template_dir = Path(template_dir)
        self.llm_client = llm_client

    def _build_template_path(self, action: str, platform_os: str, major_version: str) -> Path:
        sanitized_version = major_version.lower().replace(".", "_")
        return self.template_dir / action / f"{platform_os}__{sanitized_version}.ttp"

    def execute_ttp_parse(self, raw_data: str, template_content: str) -> list:
        parser = ttp(data=raw_data, template=template_content)
        parser.parse()
        results = parser.result()
        # Normalização do retorno padrão do TTP ([ [ {...} ] ])
        if results and isinstance(results, list):
            flat = results[0]
            if isinstance(flat, list) and len(flat) > 0:
                return flat
            if isinstance(flat, dict):
                return [flat]
        return []

    def parse_and_validate(
        self, 
        action: str, 
        platform_os: str, 
        major_version: str, 
        raw_stdout: str, 
        schema_cls: Type[SystemVersionSchema]
    ) -> dict:
        template_file = self._build_template_path(action, platform_os, major_version)

        # 1. Tentativa de Cache Local
        if template_file.exists():
            template_str = template_file.read_text(encoding="utf-8")
            parsed_records = self.execute_ttp_parse(raw_stdout, template_str)
            if parsed_records:
                try:
                    validated = schema_cls.model_validate(parsed_records[0])
                    return validated.model_dump()
                except ValidationError:
                    pass  # Template do cache corrompido ou versão alterou a sintaxe

        # 2. Loop de Auto-Cura com LLM
        return self._auto_heal_template_loop(
            action=action,
            platform_os=platform_os,
            major_version=major_version,
            raw_stdout=raw_stdout,
            schema_cls=schema_cls,
            dest_path=template_file
        )

    def _auto_heal_template_loop(
        self, 
        action: str, 
        platform_os: str, 
        major_version: str, 
        raw_stdout: str, 
        schema_cls: Type[SystemVersionSchema],
        dest_path: Path,
        max_retries: int = 3
    ) -> dict:
        sampled_raw = extract_representative_sample(raw_stdout)
        schema_json = json.dumps(schema_cls.model_json_schema(), indent=2)
        last_error = "Nenhum erro prévio."

        for attempt in range(1, max_retries + 1):
            prompt = f"""
            Você é um especialista em rede e parsing com TTP (Template Text Parser).
            Crie um template TTP rigoroso para processar o output CLI abaixo.

            OUTPUT AMOSTRADO:
            {sampled_raw}

            SCHEMA JSON ESPERADO:
            {schema_json}

            FEEDBACK DA TENTATIVA ANTERIOR:
            {last_error}

            REGRAS OBRIGATÓRIAS:
            1. Retorne APENAS o código do template TTP delimitado por ```ttp ... ```.
            2. Use variáveis com regex ou filtros do TTP: ex: {{{{ hostname }}}}, {{{{ os_version | ORPHRASE }}}}.
            3. Certifique-se de que os nomes das variáveis coincidam exatamente com os campos do schema.
            """

            # Chamada abstrata para LLM (ex: Claude 3.5 Sonnet / GPT-4o)
            generated_ttp = self.llm_client.generate(prompt)
            clean_ttp = self._extract_code_block(generated_ttp)

            # Teste no raw integral (Sandbox Local)
            parsed_data = self.execute_ttp_parse(raw_stdout, clean_ttp)
            if not parsed_data:
                last_error = f"Tentativa {attempt}: TTP parser retornou lista vazia. Variáveis não deram match."
                continue

            try:
                validated = schema_cls.model_validate(parsed_data[0])
                # Sucesso: Persiste no cache
                dest_path.parent.mkdir(parents=True, exist_ok=True)
                dest_path.write_text(clean_ttp, encoding="utf-8")
                return validated.model_dump()
            except ValidationError as ve:
                last_error = f"Tentativa {attempt}: Validação Pydantic falhou: {ve.json()}"

        raise RuntimeError(f"Falha de auto-cura após {max_retries} tentativas. Último erro: {last_error}")

    @staticmethod
    def _extract_code_block(text: str) -> str:
        if "```ttp" in text:
            return text.split("```ttp")[1].split("```")[0].strip()
        if "```" in text:
            return text.split("```")[1].split("```")[0].strip()
        return text.strip()

10. Operational Checklist for Production DeploymentRead-Only Enforced Credentials: Configure terminal users with TACACS+/RADIUS authorization restricted to view/operational privileges (e.g., privilege 1 on Cisco, monitor role on Junos/Huawei). Block configure terminal, system-view, or destructive routines.Terminal Hygiene: Ensure commands disabling pagination (terminal length 0, screen-length 0 temporary) are executed implicitly within the MCP session initialization.Template Version Control: Maintain the storage/templates/ folder as a Git repository. Auto-generated templates can trigger automated merge requests (MRs) for operator sign-off, facilitating review of auto-healed rules.Resilience to Concurrency: Store connection pools inside the MCP daemon (using Scrapli's AsyncScrapli or Netmiko connection pools) so multiple dependent commands re-use the established SSH channel.
